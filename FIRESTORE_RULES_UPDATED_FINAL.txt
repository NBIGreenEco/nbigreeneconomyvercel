rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Enhanced isAdmin() function - checks email directly (session-based verification)
    function isAdmin() {
      return request.auth != null && (
        request.auth.token.email == 'nbigreeneconomy@gmail.com' ||
        exists(/databases/$(database)/documents/admins/$(request.auth.token.email))
      );
    }

    match /admin_config/password_hash {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.token.email == 'nbigreeneconomy@gmail.com';
    }

    match /admin_config/{docId} {
      allow read, write: if request.auth != null && request.auth.token.email == 'nbigreeneconomy@gmail.com';
      allow create: if false;
    }

    match /users/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow write: if request.auth != null && (
        request.auth.uid == userId ||
        isAdmin()
      );
    }

    match /newsletterSubscriptions/{document} {
      allow read: if isAdmin();
      allow write: if request.auth != null &&
        request.resource.data.keys().hasOnly(['email', 'name', 'subscribedAt', 'status', 'source']) &&
        request.resource.data.email is string &&
        request.resource.data.subscribedAt is timestamp &&
        request.resource.data.status is string &&
        request.resource.data.source is string;
    }

    match /verification_codes/{email} {
      allow read: if request.auth != null && request.auth.token.email == email && resource.data.expiresAt > request.time;
      allow write, delete: if request.auth != null && (
        request.auth.token.email == email ||
        isAdmin()
      ) && request.resource.data.keys().hasOnly(['email', 'codeHash', 'createdAt', 'expiresAt']);
    }

    // EMAIL_VERIFICATIONS - Consolidated rules for both authenticated and unauthenticated writes
    match /email_verifications/{email} {
      allow read: if request.auth != null && request.auth.token.email == email;
      allow write: if (
        // Case 1: Admin signup - Authenticated writes with email, token, isVerified, createdAt, expiresAt
        (request.auth != null && (
          request.auth.token.email == email ||
          isAdmin()
        ) && request.resource.data.keys().hasOnly(['email', 'token', 'isVerified', 'createdAt', 'expiresAt']))
        ||
        // Case 2: Regular user signup - Authenticated writes with minimal fields (email, isVerified, createdAt)
        (request.auth != null && request.auth.token.email == email &&
         request.resource.data.keys().hasOnly(['email', 'isVerified', 'createdAt']))
        ||
        // Case 3: VerifyEmail - Unauthenticated OR Authenticated writes with verification fields
        (request.resource.data.keys().hasOnly(['email', 'isVerified', 'emailVerified', 'verifiedAt']) &&
          request.resource.data.isVerified is bool &&
          request.resource.data.emailVerified is bool &&
          request.resource.data.verifiedAt is timestamp)
      );
      allow delete: if request.auth != null && isAdmin();
    }

    // Events - Admin-only operations with audit trail
    match /events/{eventId} {
      allow read: if true;
      allow create: if (
        isAdmin() ||
        (request.resource.data.createdBy == 'nbigreeneconomy@gmail.com' &&
         request.resource.data.updatedBy == 'nbigreeneconomy@gmail.com')
      );
      allow update: if (
        isAdmin() ||
        (request.resource.data.updatedBy == 'nbigreeneconomy@gmail.com')
      );
      allow delete: if isAdmin() || resource.data.createdBy == 'nbigreeneconomy@gmail.com';
    }

    // News - Admin-only operations with audit trail
    match /news/{newsId} {
      allow read: if true;
      allow create: if (
        isAdmin() ||
        (request.resource.data.createdBy == 'nbigreeneconomy@gmail.com' &&
         request.resource.data.updatedBy == 'nbigreeneconomy@gmail.com')
      );
      allow update: if (
        isAdmin() ||
        (request.resource.data.updatedBy == 'nbigreeneconomy@gmail.com')
      );
      allow delete: if isAdmin() || resource.data.createdBy == 'nbigreeneconomy@gmail.com';
    }

    // Translations - Admin-only operations
    match /translations/{translationId} {
      allow read: if true;
      allow create: if isAdmin() && request.resource.data.keys().hasOnly([
        'language', 'key', 'value', 'namespace',
        'createdAt', 'updatedAt', 'createdBy', 'updatedBy'
      ]);
      allow update: if isAdmin() && request.resource.data.keys().hasOnly([
        'language', 'key', 'value', 'namespace',
        'createdAt', 'updatedAt', 'createdBy', 'updatedBy'
      ]);
      allow delete: if isAdmin();
    }

    match /translation_metadata/{docId} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }

    // INTERACTIONS - Updated to allow both tempUserId and userId fields for backward compatibility
    // Allow unauthenticated writes for tracking interactions during sign-up
    match /interactions/{interactionId} {
      allow read: if isAdmin();
      allow write: if (
        // Primary rule: tempUserId fields (new standard)
        (request.resource.data.keys().hasOnly(['tempUserId', 'category', 'action', 'label', 'timestamp', 'language', 'userAgent']) &&
         request.resource.data.tempUserId is string &&
         request.resource.data.category is string &&
         request.resource.data.action is string &&
         request.resource.data.label is string &&
         request.resource.data.language is string &&
         request.resource.data.userAgent is string)
        ||
        // Fallback: userId fields (legacy, for signin_clean.js compatibility)
        (request.resource.data.keys().hasOnly(['userId', 'category', 'action', 'label', 'timestamp', 'language', 'userAgent']) &&
         request.resource.data.userId is string &&
         request.resource.data.category is string &&
         request.resource.data.action is string &&
         request.resource.data.label is string &&
         request.resource.data.language is string &&
         request.resource.data.userAgent is string)
      );
    }

    match /admins/{adminEmail} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && (
        request.auth.token.email == adminEmail ||
        isAdmin()
      ) && request.resource.data.keys().hasOnly(['isAdmin']);
    }

    match /questionnaire_responses/{responseId} {
      allow read: if request.auth != null && (
        resource.data.userId == request.auth.uid ||
        isAdmin()
      );
      allow write: if request.auth != null && (
        request.auth.uid == request.resource.data.userId ||
        isAdmin()
      ) && request.resource.data.keys().hasOnly([
        'responseId', 'experienceLevel', 'preferredLanguage', 'role',
        'primaryInterest', 'fundingType', 'businessSector', 'greenTools',
        'userId', 'timestamp'
      ]);
      allow delete: if isAdmin();
    }

    // OPPORTUNITIES - Admin-only operations with audit trail
    match /opportunities/{opportunityId} {
      allow read: if true;
      allow create: if (
        isAdmin() ||
        (request.resource.data.createdBy == 'nbigreeneconomy@gmail.com' &&
         request.resource.data.updatedBy == 'nbigreeneconomy@gmail.com')
      );
      allow update: if (
        isAdmin() ||
        (request.resource.data.updatedBy == 'nbigreeneconomy@gmail.com')
      );
      allow delete: if isAdmin() || resource.data.createdBy == 'nbigreeneconomy@gmail.com';
    }

    // FUNDING - Admin-only operations with audit trail
    match /funding/{fundingId} {
      allow read: if true;
      allow create: if (
        isAdmin() ||
        (request.resource.data.createdBy == 'nbigreeneconomy@gmail.com' &&
         request.resource.data.updatedBy == 'nbigreeneconomy@gmail.com')
      );
      allow update: if (
        isAdmin() ||
        (request.resource.data.updatedBy == 'nbigreeneconomy@gmail.com')
      );
      allow delete: if isAdmin() || resource.data.createdBy == 'nbigreeneconomy@gmail.com';
    }

    match /funding_applications/{applicationId} {
      allow read: if request.auth != null && (
        resource.data.userId == request.auth.uid ||
        isAdmin()
      );
      allow write: if request.auth != null && (
        request.auth.uid == request.resource.data.userId ||
        isAdmin()
      );
      allow delete: if isAdmin();
    }

    match /projects/{projectId} {
      allow read: if request.auth != null && isAdmin();
      allow write: if request.auth != null && isAdmin();
      allow delete: if request.auth != null && isAdmin();
    }

    match /metadata/{documentId} {
      allow read: if request.auth != null && isAdmin();
      allow write: if request.auth != null && isAdmin();
      allow delete: if request.auth != null && isAdmin();
    }

    match /apis/{apiId} {
      allow read: if isAdmin();
      allow write: if isAdmin();
      allow delete: if isAdmin();
    }

    // Default deny all other operations
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
